#include <iostream>
using namespace std;
int main()
{
    int i = 0; 
    int const Taille = 5;
    
    int Tableau1D[Taille];
    int Tableau2D[Taille][Taille];
    // Remplissage tableau 1D
    cout << "Tableau 1: [";
    while (i < Taille * Taille)
    {
        Tableau1D[i] = i;
        if (i < (Taille * Taille) - 1) // pour ne pas afficher , au dernier chiffre.
            cout << Tableau1D[i] << " ,";
        else 
        {
            cout << Tableau1D[i];
        }
        i++;
    }
    cout << ']';   
   
    cout << "\nTableau2: ";
    i = 0;
    //Remplissage Tableau 2D
    for (int lig = 0; lig < Taille; lig++)
    {
        for (int col = 0; col < Taille; col++)
        {
            Tableau2D[lig][col] = Tableau1D[Taille * lig + col];
           // Tableau2D[lig][col] = Tableau1D[i];
        //    ++i;
        }
                cout << '\n';
    }
    
    for (int lig = 0; lig < Taille; ++lig)
    {
        for (int col = 0; col < Taille; ++col)
        {
            cout << Tableau2D[lig][col] << ", ";
        }
        cout << '\n';
    }
     for (int lig = 0; lig < Taille; ++lig)
    {
        int SommeLig = 0;
        int SommeCol = 0;
        for (int col = 0; col < Taille; ++col)
        {
            SommeLig +=  Tableau2D[lig][col];
            SommeCol += Tableau2D[col][lig];
        }
        cout << "Somme ligne " << lig << " : " << SommeLig << endl;
        cout << "Somme colonne " << lig << " : " << SommeCol << endl;
    
     }
    return 0;
}
